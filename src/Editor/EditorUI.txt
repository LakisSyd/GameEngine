static constexpr ImVec4 COL_SECTION    = ImVec4(0.18f, 0.18f, 0.18f, 1.0f);
static constexpr ImVec4 COL_ACCENT     = ImVec4(0.06f, 0.53f, 0.98f, 1.0f);
static constexpr ImVec4 COL_ACCENT_HOV = ImVec4(0.20f, 0.65f, 1.00f, 1.0f);
static constexpr ImVec4 COL_GREEN      = ImVec4(0.13f, 0.69f, 0.30f, 1.0f);
static constexpr ImVec4 COL_GREEN_HOV  = ImVec4(0.18f, 0.85f, 0.40f, 1.0f);

EditorUI::~EditorUI() { Shutdown(); }

bool EditorUI::Initialize(HWND hwnd, ID3D11Device* device, ID3D11DeviceContext* context)
{
    IMGUI_CHECKVERSION();
    ImGui::CreateContext();

    ImGuiIO& io = ImGui::GetIO();
    io.ConfigFlags |= ImGuiConfigFlags_NavEnableKeyboard;

    // ── Тёмная тема в стиле UE5 ──────────────────────────────────────────────
    ImGui::StyleColorsDark();
    ImGuiStyle& s = ImGui::GetStyle();

    s.WindowRounding    = 4.0f;
    s.FrameRounding     = 3.0f;
    s.PopupRounding     = 3.0f;
    s.ScrollbarRounding = 3.0f;
    s.GrabRounding      = 3.0f;
    s.WindowBorderSize  = 0.0f;
    s.FrameBorderSize   = 0.0f;
    s.ItemSpacing       = ImVec2(6, 5);
    s.FramePadding      = ImVec2(8, 4);
    s.IndentSpacing     = 16.0f;

    ImVec4* c = s.Colors;
    c[ImGuiCol_WindowBg]         = ImVec4(0.10f, 0.10f, 0.10f, 1.00f);
    c[ImGuiCol_ChildBg]          = ImVec4(0.12f, 0.12f, 0.12f, 1.00f);
    c[ImGuiCol_Header]           = ImVec4(0.16f, 0.16f, 0.16f, 1.00f);
    c[ImGuiCol_HeaderHovered]    = ImVec4(0.22f, 0.22f, 0.22f, 1.00f);
    c[ImGuiCol_HeaderActive]     = COL_ACCENT;
    c[ImGuiCol_Button]           = ImVec4(0.22f, 0.22f, 0.22f, 1.00f);
    c[ImGuiCol_ButtonHovered]    = ImVec4(0.30f, 0.30f, 0.30f, 1.00f);
    c[ImGuiCol_ButtonActive]     = COL_ACCENT;
    c[ImGuiCol_FrameBg]          = ImVec4(0.16f, 0.16f, 0.16f, 1.00f);
    c[ImGuiCol_FrameBgHovered]   = ImVec4(0.22f, 0.22f, 0.22f, 1.00f);
    c[ImGuiCol_TitleBg]          = ImVec4(0.08f, 0.08f, 0.08f, 1.00f);
    c[ImGuiCol_TitleBgActive]    = ImVec4(0.08f, 0.08f, 0.08f, 1.00f);
    c[ImGuiCol_Tab]              = ImVec4(0.14f, 0.14f, 0.14f, 1.00f);
    c[ImGuiCol_TabHovered]       = ImVec4(0.28f, 0.28f, 0.28f, 1.00f);
    c[ImGuiCol_TabActive]        = COL_ACCENT;
    c[ImGuiCol_Separator]        = ImVec4(0.25f, 0.25f, 0.25f, 1.00f);
    c[ImGuiCol_ScrollbarBg]      = ImVec4(0.10f, 0.10f, 0.10f, 1.00f);
    c[ImGuiCol_ScrollbarGrab]    = ImVec4(0.30f, 0.30f, 0.30f, 1.00f);
    c[ImGuiCol_CheckMark]        = COL_ACCENT;
    c[ImGuiCol_SliderGrab]       = COL_ACCENT;
    c[ImGuiCol_SliderGrabActive] = COL_ACCENT_HOV;

    // ── Масштабирование шрифта под разрешение ────────────────────────────────
    // 💡 ImGui рендерит шрифт в пикселях — при высоком DPI нужен крупнее
    const float dpiScale = static_cast<float>(GetDpiForWindow(hwnd)) / 96.0f;
    io.FontGlobalScale = dpiScale > 1.0f ? dpiScale : 1.0f;

    if (!ImGui_ImplWin32_Init(hwnd))           return false;
    if (!ImGui_ImplDX11_Init(device, context)) return false;
    return true;
}

void EditorUI::Shutdown()
{
    ImGui_ImplDX11_Shutdown();
    ImGui_ImplWin32_Shutdown();
    ImGui::DestroyContext();
}

void EditorUI::NewFrame()
{
    ImGui_ImplDX11_NewFrame();
    ImGui_ImplWin32_NewFrame();
    ImGui::NewFrame();
}

void EditorUI::Render()
{
    ImGui::Render();
    ImGui_ImplDX11_RenderDrawData(ImGui::GetDrawData());
}

bool EditorUI::WantCaptureMouse() const { return ImGui::GetIO().WantCaptureMouse; }

// ── Вспомогательный заголовок секции ─────────────────────────────────────────
void EditorUI::SectionHeader(const char* label)
{
    const float headerH = std::floor(24.0f * EditorUI::Scale());
    ImGui::PushStyleColor(ImGuiCol_ChildBg, ImVec4(0.16f, 0.16f, 0.16f, 1.0f));
    ImGui::BeginChild(label, ImVec2(-1, headerH), false);
    ImGui::SetCursorPosY(ImGui::GetCursorPosY() + std::floor(4.0f * EditorUI::Scale()));
    ImGui::TextColored(ImVec4(0.75f, 0.75f, 0.75f, 1.0f), "  %s", label);
    ImGui::EndChild();
    ImGui::PopStyleColor();
}

// ── XYZ DragFloat3 с цветными метками как в UE5 ──────────────────────────────
static bool DragFloat3Colored(const char* id, float v[3], float speed = 0.1f)
{
    bool changed = false;
    ImGui::PushID(id);
    const float w = (ImGui::GetContentRegionAvail().x - 6.0f) / 3.0f;

    // X
    ImGui::PushStyleColor(ImGuiCol_FrameBg, ImVec4(0.30f, 0.08f, 0.08f, 1.0f));
    ImGui::SetNextItemWidth(w);
    if (ImGui::DragFloat("##X", &v[0], speed)) changed = true;
    ImGui::PopStyleColor();
    ImGui::SameLine(0, 2);

    // Y
    ImGui::PushStyleColor(ImGuiCol_FrameBg, ImVec4(0.08f, 0.25f, 0.08f, 1.0f));
    ImGui::SetNextItemWidth(w);
    if (ImGui::DragFloat("##Y", &v[1], speed)) changed = true;
    ImGui::PopStyleColor();
    ImGui::SameLine(0, 2);

    // Z
    ImGui::PushStyleColor(ImGuiCol_FrameBg, ImVec4(0.08f, 0.13f, 0.30f, 1.0f));
    ImGui::SetNextItemWidth(w);
    if (ImGui::DragFloat("##Z", &v[2], speed)) changed = true;
    ImGui::PopStyleColor();

    ImGui::PopID();
    return changed;
}

//=============================================================================
// Update — главный метод
//=============================================================================
void EditorUI::Update(SceneManager& sceneManager)
{
    NewFrame();
    RenderSidebar(sceneManager);
    RenderScenePreview();
}

//=============================================================================
// RenderSidebar — фиксированная правая боковая панель в стиле UE5
//=============================================================================
void EditorUI::RenderSidebar(SceneManager& sceneManager)
{
    const ImGuiIO& io      = ImGui::GetIO();
    const float    height  = io.DisplaySize.y;
    const float    sidebarW = SidebarWidth();   // 💡 Теперь масштабируется
    const float    posX    = io.DisplaySize.x - sidebarW;

    ImGui::SetNextWindowPos (ImVec2(posX, 0),              ImGuiCond_Always);
    ImGui::SetNextWindowSize(ImVec2(sidebarW, height),     ImGuiCond_Always);
    ImGui::SetNextWindowBgAlpha(0.96f);

    ImGuiWindowFlags flags =
        ImGuiWindowFlags_NoTitleBar       |
        ImGuiWindowFlags_NoResize         |
        ImGuiWindowFlags_NoMove           |
        ImGuiWindowFlags_NoScrollbar      |
        ImGuiWindowFlags_NoScrollWithMouse|
        ImGuiWindowFlags_NoBringToFrontOnFocus;

    ImGui::PushStyleVar(ImGuiStyleVar_WindowPadding, ImVec2(0, 0));
    ImGui::Begin("##Sidebar", nullptr, flags);
    ImGui::PopStyleVar();

    // ── Заголовок ─────────────────────────────────────────────────────────────
    const float headerH = std::floor(36.0f * ScaleY());
    ImGui::PushStyleColor(ImGuiCol_ChildBg, ImVec4(0.07f, 0.07f, 0.07f, 1.0f));
    ImGui::BeginChild("##SidebarHeader", ImVec2(-1, headerH), false);
    ImGui::SetCursorPos(ImVec2(10, std::floor(9.0f * ScaleY())));
    ImGui::TextColored(COL_ACCENT, "◈");
    ImGui::SameLine();
    ImGui::Text("Scene Editor");
    ImGui::SameLine(sidebarW - std::floor(70.0f * ScaleX()));
    ImGui::TextColored(COL_TEXT_DIM, "%zu obj", sceneManager.GetObjectCount());
    ImGui::EndChild();
    ImGui::PopStyleColor();

    // ── Вкладки ───────────────────────────────────────────────────────────────
    ImGui::PushStyleVar(ImGuiStyleVar_ItemSpacing, ImVec2(0, 0));

    // 💡 FittingPolicyResizeDown — сжимает вкладки, если не помещаются по ширине,
    //    вместо того чтобы обрезать/уводить за границу
    const ImGuiTabBarFlags tabFlags =
        ImGuiTabBarFlags_FittingPolicyResizeDown |
        ImGuiTabBarFlags_NoTooltip;

    if (ImGui::BeginTabBar("##SidebarTabs", tabFlags))
    {
        if (ImGui::BeginTabItem("Outliner"))
        {
            ImGui::PushStyleVar(ImGuiStyleVar_WindowPadding, ImVec2(8, 6));
            RenderSceneOutliner(sceneManager);
            ImGui::PopStyleVar();
            ImGui::EndTabItem();
        }
        if (ImGui::BeginTabItem("Details"))
        {
            ImGui::PushStyleVar(ImGuiStyleVar_WindowPadding, ImVec2(8, 6));
            RenderDetailsPanel(sceneManager);
            ImGui::PopStyleVar();
            ImGui::EndTabItem();
        }
        if (ImGui::BeginTabItem("Spawn"))
        {
            ImGui::PushStyleVar(ImGuiStyleVar_WindowPadding, ImVec2(8, 6));
            RenderSpawnPanel(sceneManager);
            ImGui::PopStyleVar();
            ImGui::EndTabItem();
        }
        if (ImGui::BeginTabItem("World"))
        {
            ImGui::PushStyleVar(ImGuiStyleVar_WindowPadding, ImVec2(8, 6));
            RenderFogPanel(sceneManager);
            ImGui::PopStyleVar();
            ImGui::EndTabItem();
        }
        ImGui::EndTabBar();
    }
    ImGui::PopStyleVar();

    ImGui::End();

    // ── Оверлей-selected (левый верхний угол) ───────────────────────────────
    ImGui::SetNextWindowPos (ImVec2(10, 10), ImGuiCond_Always);
    ImGui::SetNextWindowBgAlpha(0.70f);
    ImGui::PushStyleVar(ImGuiStyleVar_WindowPadding, ImVec2(8, 6));
    if (ImGui::Begin("##Selection", nullptr,
        ImGuiWindowFlags_NoTitleBar | ImGuiWindowFlags_AlwaysAutoResize |
        ImGuiWindowFlags_NoMove     | ImGuiWindowFlags_NoResize))
    {
        switch (m_selectedType)
        {
        case SelectedType::Object:
            ImGui::TextColored(COL_ACCENT, "◈"); ImGui::SameLine();
            ImGui::Text("Object #%d", m_selectedIndex);
            break;
        case SelectedType::Light:
            ImGui::TextColored(ImVec4(1,0.9f,0.3f,1), "◉"); ImGui::SameLine();
            ImGui::Text("Light #%d", m_selectedIndex);
            break;
        case SelectedType::Sprite:
            ImGui::TextColored(ImVec4(0.3f,0.8f,1,1), "◧"); ImGui::SameLine();
            ImGui::Text("Sprite #%d", m_selectedIndex);
            break;
        default:
            ImGui::TextColored(COL_TEXT_DIM, "Nothing selected");
        }
    }
    ImGui::End();
    ImGui::PopStyleVar();
}

//=============================================================================
// RenderSceneOutliner — список всех объектов сцены
//=============================================================================
void EditorUI::RenderSceneOutliner(SceneManager& sceneManager)
{
    const float s  = ScaleY();
    const float rowH   = std::floor(20.0f * s);
    const float listObjH    = std::floor(160.0f * s);
    const float listLightH  = std::floor(110.0f * s);
    const float listSprH    = std::floor(80.0f  * s);

    ImGui::Spacing();

    // Objects
    SectionHeader("📦 Objects");
    ImGui::BeginChild("##ObjList", ImVec2(-1, listObjH), true);
    for (size_t i = 0; i < sceneManager.GetObjectCount(); ++i)
    {
        auto* obj = sceneManager.GetObject(static_cast<int>(i));
        if (!obj) continue;

        const bool sel = (m_selectedType == SelectedType::Object && m_selectedIndex == (int)i);
        if (sel)
            ImGui::PushStyleColor(ImGuiCol_Header, COL_SELECTED);

        char label[64];
        snprintf(label, sizeof(label), "  ◈  Object #%zu  (%.1f, %.1f, %.1f)",
            i, obj->GetPosition().x, obj->GetPosition().y, obj->GetPosition().z);

        if (ImGui::Selectable(label, sel, ImGuiSelectableFlags_SpanAllColumns, ImVec2(0, rowH)))
            SetSelection(SelectedType::Object, static_cast<int>(i));

        if (sel) ImGui::PopStyleColor();
    }
    ImGui::EndChild();

    ImGui::Spacing();

    // Lights
    SectionHeader("💡 Lights");
    ImGui::BeginChild("##LightList", ImVec2(-1, listLightH), true);
    for (int i = 0; i < NUM_LIGHTS; ++i)
    {
        if (!sceneManager.IsLightActive(i)) continue;
        auto pos   = sceneManager.GetLightPosition(i);
        auto color = sceneManager.GetLightColor(i);

        const bool sel = (m_selectedType == SelectedType::Light && m_selectedIndex == i);
        if (sel) ImGui::PushStyleColor(ImGuiCol_Header, COL_SELECTED);

        char label[128];
        snprintf(label, sizeof(label), "  ◉  Light #%d", i);

        if (ImGui::Selectable(label, sel, 0, ImVec2(0, rowH)))
            SetSelection(SelectedType::Light, i);

        // Маленький цветной квадратик цвета источника
        const float colorBtnSize = std::floor(14.0f * s);
        ImGui::SameLine(std::floor(160.0f * ScaleX()));
        ImGui::ColorButton("##lc", ImVec4(color.x, color.y, color.z, 1),
            ImGuiColorEditFlags_NoTooltip | ImGuiColorEditFlags_NoBorder,
            ImVec2(colorBtnSize, colorBtnSize));

        if (sel) ImGui::PopStyleColor();
    }
    ImGui::EndChild();

    ImGui::Spacing();

    // Sprites
    SectionHeader("🖼️ Sprites");
    ImGui::BeginChild("##SpriteList", ImVec2(-1, listSprH), true);
    for (size_t i = 0; i < sceneManager.GetSpriteCount(); ++i)
    {
        auto* spr = sceneManager.GetSprite(static_cast<int>(i));
        if (!spr) continue;

        const bool sel = (m_selectedType == SelectedType::Sprite && m_selectedIndex == (int)i);
        if (sel) ImGui::PushStyleColor(ImGuiCol_Header, COL_SELECTED);

        char label[64];
        auto pos = spr->GetPosition();
        snprintf(label, sizeof(label), "  ◧  Sprite #%zu", i);
        if (ImGui::Selectable(label, sel, 0, ImVec2(0, rowH)))
            SetSelection(SelectedType::Sprite, static_cast<int>(i));

        if (sel) ImGui::PopStyleColor();
    }
    ImGui::EndChild();
}

//=============================================================================
// RenderDetailsPanel — свойства выбранного объекта
//=============================================================================
void EditorUI::RenderDetailsPanel(SceneManager& sceneManager)
{
    if (m_selectedType == SelectedType::None || m_selectedIndex < 0)
    {
        ImGui::Spacing();
        ImGui::SetCursorPosX((SidebarWidth() - std::floor(160.0f * ScaleX())) * 0.5f);
        ImGui::TextColored(COL_TEXT_DIM, "Select an object in Outliner");
        return;
    }

    ImGui::Spacing();

    switch (m_selectedType)
    {
    case SelectedType::Object:
        if (auto* obj = sceneManager.GetObject(m_selectedIndex))
            RenderObjectProperties(obj);
        break;
    case SelectedType::Sprite:
        if (auto* spr = sceneManager.GetSprite(m_selectedIndex))
            RenderSpriteProperties(spr);
        break;
    case SelectedType::Light:
        RenderLightProperties(sceneManager, m_selectedIndex);
        break;
    default: break;
    }
}

//=============================================================================
// RenderObjectProperties — позиция + вращение + масштаб в стиле UE5
//=============================================================================
void EditorUI::RenderObjectProperties(GameObject* obj)
{
    // ── Transform ────────────────────────────────────────────────────────────
    SectionHeader("Transform");
    ImGui::Spacing();

    // Позиция
    ImGui::TextColored(COL_TEXT_DIM, " Location");
    float pos[3] = { obj->GetPosition().x, obj->GetPosition().y, obj->GetPosition().z };
    if (DragFloat3Colored("##Pos", pos, 0.1f))
        obj->SetPosition(pos[0], pos[1], pos[2]);

    ImGui::Spacing();

    // Вращение
    ImGui::TextColored(COL_TEXT_DIM, " Rotation");
    float rot[3] = {
        XMConvertToDegrees(obj->GetRotation().x),
        XMConvertToDegrees(obj->GetRotation().y),
        XMConvertToDegrees(obj->GetRotation().z)
    };
    if (DragFloat3Colored("##Rot", rot, 0.5f))
        obj->SetRotation(
            XMConvertToRadians(rot[0]),
            XMConvertToRadians(rot[1]),
            XMConvertToRadians(rot[2]));

    ImGui::Spacing();

    // Масштаб
    ImGui::TextColored(COL_TEXT_DIM, " Scale");
    float scl[3] = { obj->GetScale().x, obj->GetScale().y, obj->GetScale().z };

    // Кнопка равномерного масштаба
    static bool uniformScale = true;
    ImGui::SameLine(SidebarWidth() - std::floor(80.0f * ScaleX()));
    ImGui::Checkbox("Uniform", &uniformScale);

    if (uniformScale)
    {
        ImGui::SetNextItemWidth(-1);
        float avg = (scl[0] + scl[1] + scl[2]) / 3.0f;
        if (ImGui::DragFloat("##SclU", &avg, 0.01f, 0.001f, 100.0f))
            obj->SetScale(avg, avg, avg);
    }
    else
    {
        if (DragFloat3Colored("##Scl", scl, 0.01f))
            obj->SetScale(scl[0], scl[1], scl[2]);
    }

    ImGui::Spacing();

    // Кнопка сброса
    ImGui::PushStyleColor(ImGuiCol_Button,        ImVec4(0.20f, 0.20f, 0.20f, 1));
    ImGui::PushStyleColor(ImGuiCol_ButtonHovered, ImVec4(0.30f, 0.30f, 0.30f, 1));
    if (ImGui::Button("Reset Transform", ImVec2(-1, 0)))
    {
        obj->SetPosition(0, 0, 0);
        obj->SetRotation(0, 0, 0);
        obj->SetScale(1, 1, 1);
    }
    ImGui::PopStyleColor(2);
}

//=============================================================================
// RenderSpriteProperties
//=============================================================================
void EditorUI::RenderSpriteProperties(WorldSpriteClass* sprite)
{
    SectionHeader("Transform");
    ImGui::Spacing();

    ImGui::TextColored(COL_TEXT_DIM, " Location");
    XMFLOAT3 pos = sprite->GetPosition();
    float posArr[3] = { pos.x, pos.y, pos.z };
    if (DragFloat3Colored("##SprPos", posArr, 0.1f))
        sprite->SetPosition(posArr[0], posArr[1], posArr[2]);

    ImGui::Spacing();
    ImGui::TextColored(COL_TEXT_DIM, " Scale");
    XMFLOAT2 scl = sprite->GetScale();
    float sclArr[2] = { scl.x, scl.y };
    ImGui::PushItemWidth(-1);
    if (ImGui::DragFloat2("##SprScl", sclArr, 0.01f, 0.01f, 10.0f))
        sprite->SetScale(sclArr[0], sclArr[1]);
    ImGui::PopItemWidth();
}

//=============================================================================
// RenderLightProperties
//=============================================================================
void EditorUI::RenderLightProperties(SceneManager& sceneManager, int idx)
{
    SectionHeader("Light");
    ImGui::Spacing();

    XMFLOAT4 pos4   = sceneManager.GetLightPosition(idx);
    XMFLOAT4 color4 = sceneManager.GetLightColor(idx);

    ImGui::TextColored(COL_TEXT_DIM, " Position");
    float posArr[3] = { pos4.x, pos4.y, pos4.z };
    if (DragFloat3Colored("##LPos", posArr, 0.1f))
    {
        XMFLOAT3 p(posArr[0], posArr[1], posArr[2]);
        sceneManager.SetLight(idx, p, color4, true);
    }

    ImGui::Spacing();
    ImGui::TextColored(COL_TEXT_DIM, " Color");
    XMFLOAT3 rgb = { color4.x, color4.y, color4.z };
    ImGui::PushItemWidth(-1);
    if (ImGui::ColorEdit3("##LCol", &rgb.x))
    {
        XMFLOAT4 nc(rgb.x, rgb.y, rgb.z, color4.w);
        XMFLOAT3 p(pos4.x, pos4.y, pos4.z);
        sceneManager.SetLight(idx, p, nc, true);
    }
    ImGui::PopItemWidth();

    ImGui::Spacing();
    bool active = sceneManager.IsLightActive(idx);
    if (ImGui::Checkbox("Active", &active))
    {
        XMFLOAT3 p(pos4.x, pos4.y, pos4.z);
        sceneManager.SetLight(idx, p, color4, active);
    }
    ImGui::SameLine();
    ImGui::PushStyleColor(ImGuiCol_Button,        COL_RED);
    ImGui::PushStyleColor(ImGuiCol_ButtonHovered, ImVec4(0.9f, 0.3f, 0.3f, 1.0f));
    if (ImGui::Button("Detach")) sceneManager.DetachLight(idx);
    ImGui::PopStyleColor(2);
}

//=============================================================================
// RenderSpawnPanel
//=============================================================================
void EditorUI::RenderSpawnPanel(SceneManager& sceneManager)
{
    ImGui::Spacing();

    // ── Модель ───────────────────────────────────────────────────────────────
    SectionHeader("Model");
    ImGui::Spacing();
    
    const char* modelLabels[PRESET_COUNT] = { "Sphere", "Cube", "Plane", "Cylinder" };
    ImGui::PushStyleColor(ImGuiCol_Button,        ImVec4(0.17f, 0.22f, 0.35f, 1));
    ImGui::PushStyleColor(ImGuiCol_ButtonHovered, COL_ACCENT);
    const float bw = (ImGui::GetContentRegionAvail().x - 6.0f) / PRESET_COUNT;
    for (int i = 0; i < PRESET_COUNT; i++)
    {
        if (i > 0) ImGui::SameLine(0, 2);
        if (ImGui::Button(modelLabels[i], ImVec2(bw, 0)))
            strncpy_s(m_spawnModelPath, sizeof(m_spawnModelPath), MODEL_PRESETS[i], _TRUNCATE);
    }
    ImGui::PopStyleColor(2);
    ImGui::Spacing();
    ImGui::SetNextItemWidth(-1);
    ImGui::InputText("##ModelPath", m_spawnModelPath, sizeof(m_spawnModelPath));
    ImGui::TextColored(COL_TEXT_DIM, "  Model path");

    ImGui::Spacing();

    // ── Текстура ─────────────────────────────────────────────────────────────
    SectionHeader("Texture");
    ImGui::Spacing();

    const char* texLabels[PRESET_COUNT] = { "Stone", "Dirt", "UV", "Arrow" };
    ImGui::PushStyleColor(ImGuiCol_Button,        ImVec4(0.30f, 0.18f, 0.10f, 1));
    ImGui::PushStyleColor(ImGuiCol_ButtonHovered, ImVec4(0.55f, 0.30f, 0.15f, 1));
    for (int i = 0; i < PRESET_COUNT; i++)
    {
        if (i > 0) ImGui::SameLine(0, 2);
        if (ImGui::Button(texLabels[i], ImVec2(bw, 0)))
            strncpy_s(m_spawnTexturePath, sizeof(m_spawnTexturePath), TEXTURE_PRESETS[i], _TRUNCATE);
    }
    ImGui::PopStyleColor(2);
    ImGui::Spacing();
    ImGui::SetNextItemWidth(-1);
    ImGui::InputText("##TexPath", m_spawnTexturePath, sizeof(m_spawnTexturePath));
    ImGui::TextColored(COL_TEXT_DIM, "  Texture path");

    ImGui::Spacing();

    // ── Позиция ───────────────────────────────────────────────────────────────
    SectionHeader("Spawn Location");
    ImGui::Spacing();
    DragFloat3Colored("##SpawnPos", m_spawnPos, 0.1f);
    ImGui::Spacing();

    ImGui::PushStyleColor(ImGuiCol_Button,        ImVec4(0.18f, 0.18f, 0.18f, 1));
    ImGui::PushStyleColor(ImGuiCol_ButtonHovered, ImVec4(0.28f, 0.28f, 0.28f, 1));
    if (ImGui::Button("Origin##rst", ImVec2((ImGui::GetContentRegionAvail().x - 2) * 0.5f, 0)))
        m_spawnPos[0] = m_spawnPos[1] = m_spawnPos[2] = 0.0f;
    ImGui::SameLine(0, 2);
    if (ImGui::Button("Random XZ", ImVec2(-1, 0)))
    {
        m_spawnPos[0] = (float)(rand() % 20 - 10);
        m_spawnPos[2] = (float)(rand() % 20 - 10);
    }
    ImGui::PopStyleColor(2);

    ImGui::Spacing();
    ImGui::Separator();
    ImGui::Spacing();

    // ── Кнопка Spawn ─────────────────────────────────────────────────────────
    const bool valid = m_spawnModelPath[0] != '\0' && m_spawnTexturePath[0] != '\0';
    if (!valid) ImGui::BeginDisabled();

    const float spawnBtnH = std::floor(38.0f * ScaleY());
    ImGui::PushStyleColor(ImGuiCol_Button,       COL_GREEN);
    ImGui::PushStyleColor(ImGuiCol_ButtonHovered, COL_GREEN_HOV);
    ImGui::PushStyleColor(ImGuiCol_ButtonActive, ImVec4(0.08f, 0.45f, 0.18f, 1));
    if (ImGui::Button("  ＋  Spawn Object", ImVec2(-1, spawnBtnH)))
    {
        m_spawnRequest.pending     = true;
        m_spawnRequest.modelPath   = m_spawnModelPath;
        m_spawnRequest.texturePath = m_spawnTexturePath;
        m_spawnRequest.posX        = m_spawnPos[0];
        m_spawnRequest.posY        = m_spawnPos[1];
        m_spawnRequest.posZ        = m_spawnPos[2];
    }
    ImGui::PopStyleColor(3);

    if (!valid)
    {
        ImGui::EndDisabled();
        ImGui::Spacing();
        ImGui::TextColored(ImVec4(1, 0.4f, 0.4f, 1), "  ⚠  Fill paths first");
    }

    ImGui::Spacing();
    ImGui::TextColored(COL_TEXT_DIM, "  Scene: %zu objects", sceneManager.GetObjectCount());
}

//=============================================================================
// RenderFogPanel
//=============================================================================
void EditorUI::RenderFogPanel(SceneManager& sceneManager)
{
    Settings::FogSettings& fog = sceneManager.GetFogSettings();

    ImGui::Spacing();
    SectionHeader("Atmosphere");
    ImGui::Spacing();

    ImGui::Checkbox("Enable Fog", &fog.enabled);
    ImGui::Spacing();

    if (fog.enabled)
    {
        ImGui::TextColored(COL_TEXT_DIM, " Start Distance");
        ImGui::SetNextItemWidth(-1);
        ImGui::SliderFloat("##FogS", &fog.FOG_START, 0.0f, 100.0f);

        ImGui::TextColored(COL_TEXT_DIM, " End Distance");
        ImGui::SetNextItemWidth(-1);
        ImGui::SliderFloat("##FogE", &fog.FOG_END, 1.0f, 500.0f);

        ImGui::Spacing();
        ImGui::TextColored(COL_TEXT_DIM, " Fog Color");
        ImGui::SetNextItemWidth(-1);
        ImGui::ColorEdit3("##FogCol", &fog.color.x);
    }
    else
    {
        ImGui::TextColored(COL_TEXT_DIM, "  Fog is disabled");
    }
}

//=============================================================================
// Scene Preview
//=============================================================================
void EditorUI::SetScenePreviewTexture(ID3D11ShaderResourceView* srv) { m_scenePreviewSRV = srv; }

void EditorUI::RenderScenePreview()
{
    if (!m_scenePreviewSRV) return;

    const ImGuiIO& io = ImGui::GetIO();

    // 💡 Размеры превью масштабируются относительно базового разрешения
    const float previewW = std::floor(io.DisplaySize.x * PREVIEW_WIDTH_RATIO);
    const float previewH = std::floor(io.DisplaySize.y * PREVIEW_HEIGHT_RATIO);
    const float imageH   = std::floor(previewH - std::floor(40.0f * ScaleY()));

    ImGui::SetNextWindowPos (ImVec2(0, io.DisplaySize.y - previewH), ImGuiCond_Always);
    ImGui::SetNextWindowSize(ImVec2(previewW, previewH),              ImGuiCond_Always);
    ImGui::SetNextWindowBgAlpha(0.88f);

    ImGuiWindowFlags flags =
        ImGuiWindowFlags_NoResize | ImGuiWindowFlags_NoMove |
        ImGuiWindowFlags_NoScrollbar | ImGuiWindowFlags_NoBringToFrontOnFocus;

    if (ImGui::Begin("Scene Preview", nullptr, flags))
    {
        ImGui::Image(m_scenePreviewSRV, ImVec2(previewW - std::floor(16.0f * ScaleX()), imageH),
            ImVec2(0, 1), ImVec2(1, 0));
    }
    ImGui::End();
}
